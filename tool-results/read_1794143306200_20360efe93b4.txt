   377→function PurchaseDialog({
   378→  products, onDone, open, setOpen,
   379→}: {
   380→  products: Product[]
   381→  onDone: () => void
   382→  open: boolean
   383→  setOpen: (v: boolean) => void
   384→}) {
   385→  const [supplier, setSupplier] = useState('')
   386→  const [items, setItems] = useState<{ productId: string; qty: string }[]>([{ productId: '', qty: '1' }])
   387→  const [saving, setSaving] = useState(false)
   388→
   389→  const submit = async () => {
   390→    const rows = items.filter((i) => i.productId && Number(i.qty) > 0).map((i) => ({ productId: i.productId, qty: Number(i.qty) }))
   391→    if (!supplier.trim() || rows.length === 0) return
   392→    setSaving(true)
   393→    const r = await fetch('/api/inventory', {
   394→      method: 'POST',
   395→      headers: { 'Content-Type': 'application/json' },
   396→      body: JSON.stringify({ supplier, items: rows }),
   397→    })
   398→    setSaving(false)
   399→    if (r.ok) {
   400→      setOpen(false)
   401→      setSupplier('')
   402→      setItems([{ productId: '', qty: '1' }])
   403→      onDone()
   404→    }
   405→  }
   406→
   407→  return (
   408→    <Dialog open={open} onOpenChange={setOpen}>
   409→      <DialogTrigger asChild>
   410→        <Button size="sm" className="rounded-xl">
   411→          <Plus className="h-4 w-4 ml-1" /> ثبت خرید
   412→        </Button>
   413→      </DialogTrigger>
   414→      <DialogContent className="rounded-2xl max-w-lg">
   415→        <DialogHeader>
   416→          <DialogTitle className="text-right">ثبت خرید انبار</DialogTitle>
   417→        </DialogHeader>
   418→        <div className="space-y-3">
   419→          <div>
   420→            <Label className="text-xs text-muted-foreground">تأمین‌کننده</Label>
   421→            <Input value={supplier} onChange={(e) => setSupplier(e.target.value)} placeholder="مثلاً: آریا دنت تجارت" className="rounded-xl mt-1" />
   422→          </div>
   423→          {items.map((it, idx) => (
   424→            <div key={idx} className="grid grid-cols-[1fr_100px] gap-2">
   425→              <Select value={it.productId} onValueChange={(v) => setItems((arr) => arr.map((x, i) => (i === idx ? { ...x, productId: v } : x)))}>
   426→                <SelectTrigger className="rounded-xl"><SelectValue placeholder="انتخاب کالا" /></SelectTrigger>
   427→                <SelectContent>
   428→                  {products.map((p) => (
   429→                    <SelectItem key={p.id} value={p.id}>{p.name}</SelectItem>
   430→                  ))}
   431→                </SelectContent>
   432→              </Select>
   433→              <Input

   531→function AddExpenseDialog({
   532→  onDone, open, setOpen,
   533→}: {
   534→  onDone: () => void
   535→  open: boolean
   536→  setOpen: (v: boolean) => void
   537→}) {
   538→  const [category, setCategory] = useState('RENT')
   539→  const [amount, setAmount] = useState('')
   540→  const [desc, setDesc] = useState('')
   541→  const [saving, setSaving] = useState(false)
   542→
   543→  const submit = async () => {
   544→    if (!amount || Number(amount) <= 0) return
   545→    setSaving(true)
   546→    const r = await fetch('/api/expenses', {
   547→      method: 'POST',
   548→      headers: { 'Content-Type': 'application/json' },
   549→      body: JSON.stringify({ category, amount: Number(amount), description: desc || 'ثبت دستی' }),
   550→    })
   551→    setSaving(false)
   552→    if (r.ok) {
   553→      setOpen(false)
   554→      setAmount('')
   555→      setDesc('')
   556→      onDone()
   557→    }
   558→  }
   559→
   560→  return (
   561→    <Dialog open={open} onOpenChange={setOpen}>
   562→      <DialogTrigger asChild>
   563→        <Button size="sm" className="rounded-xl">
   564→          <Plus className="h-4 w-4 ml-1" /> ثبت هزینه
   565→        </Button>
   566→      </DialogTrigger>
   567→      <DialogContent className="rounded-2xl max-w-md">
   568→        <DialogHeader>
   569→          <DialogTitle className="text-right">ثبت هزینه</DialogTitle>
   570→        </DialogHeader>
   571→        <div className="space-y-3">
   572→          <div>
   573→            <Label className="text-xs text-muted-foreground">دسته‌بندی</Label>
   574→            <Select value={category} onValueChange={setCategory}>
   575→              <SelectTrigger className="rounded-xl mt-1 w-full"><SelectValue /></SelectTrigger>
   576→              <SelectContent>
   577→                {['RENT', 'SALARY', 'UTILITIES', 'MARKETING', 'EQUIPMENT', 'OTHER'].map((c) => (
   578→                  <SelectItem key={c} value={c}>{EXPENSE_CATEGORY_LABEL[c]}</SelectItem>
   579→                ))}
   580→              </SelectContent>
   581→            </Select>
   582→          </div>
   583→          <div>
   584→            <Label className="text-xs text-muted-foreground">مبلغ (تومان)</Label>
   585→            <Input type="number" value={amount} onChange={(e) => setAmount(e.target.value)} className="rounded-xl mt-1" placeholder="مثلاً 45000000" />
   586→          </div>
   587→          <div>
   588→            <Label className="text-xs text-muted-foreground">شرح</Label>
   589→            <Input value={desc} onChange={(e) => setDesc(e.target.value)} className="rounded-xl mt-1" placeholder="اختیاری" />
   590→          </div>
   591→          <Button className="w-full rounded-xl" onClick={submit} disabled={saving}>
   592→            {saving ? 'در حال ثبت…' : 'ثبت هزینه'}
   593→          </Button>
   594→        </div>
   595→      </DialogContent>
   596→    </Dialog>
   597→  )
   598→}
